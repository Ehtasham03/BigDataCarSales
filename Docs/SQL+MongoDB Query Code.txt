      manufacturer: "$_id.manufacturer",
      yearOfManufacture: "$_id.year",

    }

    COUNT(a.AccidentID) AS AccidentCount
FROM Cars c
JOIN AccidentRecords a ON a.CarID = c.CarID
GROUP BY c.CarID, c.Model, c.DealerID
HAVING COUNT(a.AccidentID) > 2;

MongoDB

  {
    $addFields: {
      accidentCount: { $size: { $ifNull: ["$accidentHistory", []] } }

  {
    $project: {
      _id: 0,
      CarID: 1,
      Model: 1,
      "dealer.dealerId": 1,
      accidentCount: 1
    }
  }

SELECT 
    c.FuelType,
    CASE
        WHEN CAST(c.EngineSize AS DECIMAL(3,1)) < 1.5 THEN '<1.5L'
        WHEN CAST(c.EngineSize AS DECIMAL(3,1)) BETWEEN 1.5 AND 2.5 THEN '1.5–2.5L'
        ELSE '>2.5L'
    END AS EngineCategory,
    AVG(c.Mileage) AS AvgMileage
FROM Cars c
GROUP BY c.FuelType,
         CASE
            WHEN CAST(c.EngineSize AS DECIMAL(3,1)) < 1.5 THEN '<1.5L'
            WHEN CAST(c.EngineSize AS DECIMAL(3,1)) BETWEEN 1.5 AND 2.5 THEN '1.5–2.5L'
            ELSE '>2.5L'
         END
ORDER BY c.FuelType, EngineCategory;

MongoDB
[
  {
    $addFields: {
      engineLiters: { $toDouble: "$EngineSize" }
    }
  },
  {
    $addFields: {
      engineCategory: {
        $switch: {
          branches: [
            { case: { $lt: ["$engineLiters", 1.5] }, then: "<1.5L" },
            {
              case: {
                $and: [
                  { $gte: ["$engineLiters", 1.5] },
                  { $lte: ["$engineLiters", 2.5] }
                ]
              },
              then: "1.5–2.5L"
            }
          ],
          default: ">2.5L"
        }
      }
    }
  },
  {
    $group: {
      _id: { fuelType: "$FuelType", engineCategory: "$engineCategory" },
      avgMileage: { $avg: "$Mileage" }
    }
  },
  {
    $project: {
      _id: 0,
      fuelType: "$_id.fuelType",
      engineCategory: "$_id.engineCategory",
      avgMileage: 1
    }
  },
  { $sort: { fuelType: 1, engineCategory: 1 } }
]

WITH ServiceCounts AS (
    SELECT 
        CarID,
        COUNT(*) AS ServiceCount
    FROM ServiceRecords
    GROUP BY CarID
)
SELECT 
    c.CarID,
    c.Model,
    c.YearOfManufacture,
    sc.ServiceCount
FROM Cars c
JOIN ServiceCounts sc ON sc.CarID = c.CarID
WHERE c.YearOfManufacture <= YEAR(CURDATE()) - 10
  AND sc.ServiceCount > 2;

MongoDB
[
  {
    $addFields: {
      serviceCount: { $size: { $ifNull: ["$serviceHistory", []] } }
    }
  },
  {
    $match: {
      YearOfManufacture: { $lte: 2015 },    // FOR CURRENT YEAR = 2023
      serviceCount: { $gt: 2 }
    }
  },
  {
    $project: {
      _id: 0,
      CarID: 1,
      Model: 1,
      YearOfManufacture: 1,
      serviceCount: 1
    }
  }
]

SQL
WITH TotalCars AS (
    SELECT 
        DealerID,
        COUNT(*) AS TotalCars
    FROM Cars
    GROUP BY DealerID
),
AccidentCars AS (
    SELECT 
        c.DealerID,
        COUNT(DISTINCT c.CarID) AS AccidentProneCars
    FROM Cars c
    JOIN AccidentRecords a ON a.CarID = c.CarID
    GROUP BY c.DealerID
)
SELECT 
    d.DealerID,
    d.DealerName,
    tc.TotalCars,
    COALESCE(ac.AccidentProneCars, 0) AS AccidentProneCars,
    COALESCE(ac.AccidentProneCars, 0) / tc.TotalCars AS AccidentRatio
FROM Dealers d
JOIN TotalCars tc ON tc.DealerID = d.DealerID
LEFT JOIN AccidentCars ac ON ac.DealerID = d.DealerID
ORDER BY AccidentRatio DESC
LIMIT 3;

MongoDB
[
  {
    $addFields: {
      hasAccident: {
        $gt: [
          { $size: { $ifNull: ["$accidentHistory", []] } },
          0
        ]
      }
    }
  },
  {
    $group: {
      _id: {
        dealerId: "$dealer.dealerId",
        dealerName: "$dealer.name"
      },
      totalCars: { $sum: 1 },
      accidentProneCars: {
        $sum: {
          $cond: [{ $eq: ["$hasAccident", true] }, 1, 0]
        }
      }
    }
  },
  {
    $addFields: {
      accidentRatio: {
        $cond: [
          { $gt: ["$totalCars", 0] },
          { $divide: ["$accidentProneCars", "$totalCars"] },
          0
        ]
      }
    }
  },
  {
    $project: {
      _id: 0,
      dealerId: "$_id.dealerId",
      dealerName: "$_id.dealerName",
      totalCars: 1,
      accidentProneCars: 1,
      accidentRatio: 1
    }
  },
  { $sort: { accidentRatio: -1 } },
  { $limit: 3 }
]

WITH ManufacturerSales AS (
    SELECT 
        ManufacturerID,
        SUM(Price) AS TotalSales
    FROM Cars
    GROUP BY ManufacturerID
),
ManufacturerRepairs AS (
    SELECT 
        c.ManufacturerID,
        SUM(a.RepairCost) AS TotalRepairs
    FROM Cars c
    JOIN AccidentRecords a ON a.CarID = c.CarID
    GROUP BY c.ManufacturerID
)
SELECT 
    m.ManufacturerID,
    m.Manufacturer,
    ms.TotalSales,
    COALESCE(mr.TotalRepairs, 0) AS TotalRepairs,
    ms.TotalSales - COALESCE(mr.TotalRepairs, 0) AS NetProfit
FROM Manufacturers m
JOIN ManufacturerSales ms ON ms.ManufacturerID = m.ManufacturerID
LEFT JOIN ManufacturerRepairs mr ON mr.ManufacturerID = m.ManufacturerID
ORDER BY NetProfit DESC
LIMIT 1;

    YEAR(Date) AS ServiceYear,
    COUNT(*) AS NumServices
FROM ServiceRecords
WHERE YEAR(Date) >= YEAR(CURDATE()) - 4
GROUP BY YEAR(Date)
ORDER BY ServiceYear;

MongoDB
[
  { 
    $unwind: "$serviceHistory" 
  },
  {
    $match: {
      "serviceHistory.date": { $gte: "2020-01-01" }
    }
  },
  {
    $group: {
      _id: {
        year: { $substrBytes: ["$serviceHistory.date", 0, 4] }
      },
      numServices: { $sum: 1 }
    }
  },
  {
    $project: {
      _id: 0,
      year: "$_id.year",
      numServices: 1
    }
  },
  { 
    $sort: { year: 1 } 
  }

SET @cutoff_date = (CURRENT_DATE - INTERVAL 24 MONTH);

SELECT 
    c.CarID,
    c.Model
FROM Cars c
WHERE EXISTS (
    SELECT 1
    FROM AccidentRecords a
    WHERE a.CarID = c.CarID
      AND a.Date >= @cutoff_date
)
AND NOT EXISTS (
    SELECT 1
    FROM ServiceRecords s
    WHERE s.CarID = c.CarID
      AND s.Date >= @cutoff_date
);

MongoDB
[
  {
    $addFields: {
      recentServices: {
        $filter: {
          input: { $ifNull: ["$serviceHistory", []] },
          as: "s",
          cond: { $gte: ["$$s.date", "2023-01-01"] }
        }
      },
      recentAccidents: {
        $filter: {
          input: { $ifNull: ["$accidentHistory", []] },
          as: "a",
          cond: { $gte: ["$$a.date", "2023-01-01"] }
        }
      }
    }
  },
  {
    $match: {
      recentAccidents: { $ne: [] },
      recentServices: { $eq: [] }
    }
  },
  {
    $project: {
      _id: 0,
      CarID: 1,
      Model: 1
    }
  }
]

  },
  {
    $project: {
      _id: 0,
      manufacturerId: "$_id.manufacturerId",
      manufacturer: "$_id.manufacturer",
      severity: "$_id.severity",
      numAccidents: 1
    }

SELECT 
    f.FeatureName,
    COUNT(*) AS NumCarsWithFeature
FROM Cars c
JOIN CarFeatures cf ON cf.CarID = c.CarID
JOIN Features f ON f.FeatureID = cf.FeatureID
WHERE c.Price > 25000
GROUP BY f.FeatureName
ORDER BY NumCarsWithFeature DESC;

MongoDB
[
  { $match: { Price: { $gt: 25000 } } },
  { $unwind: "$features" },
  {
    $group: {
      _id: "$features",
      numCarsWithFeature: { $sum: 1 }
    }
  },
  {
    $project: {
      _id: 0,
      feature: "$_id",
      numCarsWithFeature: 1
    }
  },
  { $sort: { numCarsWithFeature: -1 } }
